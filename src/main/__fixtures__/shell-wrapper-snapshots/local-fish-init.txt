function __orca_shell_ready_marker --on-event fish_prompt
  builtin printf "\033]777;orca-shell-ready\007"
  functions -e __orca_shell_ready_marker
end
function __orca_codex_interactive
  for arg in $argv
    switch $arg
      case --no-daemon '--no-daemon=*' --remote '--remote=*'
        return 1
    end
  end
  while test (count $argv) -gt 0
    switch $argv[1]
      case --
        return 0
      case --no-daemon --help -h --version -V
        return 1
      case -c --config -m --model -p --profile -s --sandbox -a --ask-for-approval -C --cd --add-dir -i --image --enable --disable
        test (count $argv) -gt 1; or return 1
        set -e argv[1..2]
      case '-c=*' '--config=*' '-m=*' '--model=*' '-p=*' '--profile=*' '-s=*' '--sandbox=*' '-a=*' '--ask-for-approval=*' '-C=*' '--cd=*' '--add-dir=*' '-i=*' '--image=*' '--enable=*' '--disable=*' --search --full-auto --dangerously-bypass-approvals-and-sandbox --yolo --no-alt-screen --oss
        set -e argv[1]
      case '-*' exec e review login logout mcp mcp-server app-server completion sandbox debug apply a cloud features help agents tcp-tunnel plugin remote-control app update doctor execpolicy queue archive delete migrate-rollouts unarchive cloud-tasks responses-api-proxy stdio-to-uds exec-server
        return 1
      case '*'
        return 0
    end
  end
  return 0
end
function __orca_codex_supports_no_daemon
  command $argv[1] --no-daemon --version </dev/null >/dev/null 2>&1
end
# Why captured: an unquoted (type -t codex) expands to zero words when codex is
# absent, leaving "test = file" — fish then errors instead of failing closed.
# Quoting in place is not the fix; fish never substitutes inside double quotes.
set -l __orca_codex_type (type -t codex 2>/dev/null)
if test -x "$ORCA_CODEX_LAUNCH_PREFLIGHT"; and test "$__orca_codex_type" = file
  function codex
    if test -n "$ORCA_CODEX_HOME"
      command "$ORCA_CODEX_LAUNCH_PREFLIGHT" agent hooks prepare-codex >/dev/null 2>&1; or true
    end
    set -l executable (command -v codex 2>/dev/null)
    if test -n "$executable"; and __orca_codex_interactive $argv; and __orca_codex_supports_no_daemon "$executable"
      command "$executable" --no-daemon $argv
      return $status
    end
    command codex $argv
  end
end
set -e __orca_codex_type
